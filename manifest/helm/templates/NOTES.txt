1. Get the application URL by running these commands:
{{- if .Values.ingress.enabled }}
{{- range $host := .Values.ingress.hosts }}
  https://{{ $host.host }}
{{- end }}
{{- else if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "udrive.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo https://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.service.type }}
     NOTE: It may take a few minutes for the LoadBalancer IP to be available.
           You can watch the status of by running 'kubectl get --namespace {{ .Release.Namespace }} svc -w {{ include "udrive.fullname" . }}'
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "udrive.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo https://$SERVICE_IP:{{ .Values.service.port }}
{{- else if contains "ClusterIP" .Values.service.type }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "{{ include "udrive.selectorLabels" . }}" -o jsonpath="{.items[0].metadata.name}")
  echo "Visit https://127.0.0.1:9200 to use your application"
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 9200:9200
{{- end }}

2. Default administrator credentials:
  Username: {{ .Values.udrive.admin.username }}
{{- if .Values.udrive.admin.existingSecret }}
  Password can be fetched from secret '{{ .Values.udrive.admin.existingSecret }}' (key: '{{ .Values.udrive.admin.passwordKey }}'):
  kubectl get secret --namespace {{ .Release.Namespace }} {{ .Values.udrive.admin.existingSecret }} -o jsonpath="{.data.{{ .Values.udrive.admin.passwordKey }}}" | base64 --decode; echo
{{- else }}
  Password can be fetched from the generated secret:
  kubectl get secret --namespace {{ .Release.Namespace }} {{ include "udrive.fullname" . }} -o jsonpath="{.data.admin-password}" | base64 --decode; echo
{{- end }}

3. Check deployment status:
  kubectl rollout status deployment/{{ include "udrive.fullname" . }} -n {{ .Release.Namespace }}
